import fs from 'node:fs/promises';import path from 'node:path';import os from 'node:os';import assert from 'node:assert/strict';import {createHash,randomUUID} from 'node:crypto';import {snapshotPackage} from '../../plugin-src/src/lib/package-snapshot.mjs';
const out=path.resolve('docs/verification/production/2026-10-20-empty-lock-0328');const observed=JSON.parse(await fs.readFile(path.join(out,'observation.json'),'utf8'));const root=observed.binding.root, lock=path.join(root,'系统（别动）/库锁');
const digest=x=>createHash('sha256').update(JSON.stringify(x)).digest('hex');const same=(s,id)=>assert.ok(s.dev===id.dev&&s.ino===id.ino);
same(await fs.lstat(root),observed.root);same(await fs.lstat(lock),observed.lock);assert.deepEqual(await fs.readdir(lock),[]);assert.equal(JSON.parse(await fs.readFile(path.join(root,'内容库.json'),'utf8')).libraryId,observed.binding.libraryId);
const started=new Date().toISOString();const before=await snapshotPackage(root);assert.equal(digest(before),observed.librarySnapshot.sha256);
const dir=path.join(os.homedir(),'Library/Application Support/Nova 内容库恢复留档',`2026-10-20-empty-lock-${randomUUID()}`);await fs.mkdir(dir,{recursive:true,mode:0o700});const copy=path.join(dir,'原始内容库完整副本');await fs.cp(root,copy,{recursive:true,errorOnExist:true,force:false,preserveTimestamps:true,verbatimSymlinks:true});await fs.chmod(copy,0o700);
const [after,copied]=await Promise.all([snapshotPackage(root),snapshotPackage(copy)]);assert.deepEqual(after,before);assert.deepEqual(copied,before);assert.notEqual((await fs.lstat(copy)).ino,(await fs.lstat(root)).ino);assert.deepEqual(await fs.readdir(path.join(copy,'系统（别动）/库锁')),[]);same(await fs.lstat(lock),observed.lock);
const receipt={started,finished:new Date().toISOString(),recoveryDirectory:dir,copy,entries:Object.keys(before).length,sha256:digest(before),sourceUnchanged:true,independentCopyExact:true,emptyLockIncluded:true,sourceLock:observed.lock,scope:'Full independent preservation; source only read, no lock removal or library binding change.'};await fs.writeFile(path.join(out,'preservation.json'),JSON.stringify(receipt,null,2)+'\n');console.log(JSON.stringify(receipt));
